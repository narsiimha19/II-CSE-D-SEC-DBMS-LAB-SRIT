# (1b) 1.Implement the tables using constraints
```
CREATE TABLE STUDENT
(
  NAME VARCHAR2(40),
  STUDENT_NUMBER NUMBER PRIMARY KEY,
  CLASS NUMBER,
  MAJOR VARCHAR2(20) NOT NULL
);

CREATE TABLE COURSE 
(
  COURSE_NAME VARCHAR2(40),
  COURSE NUMBER VARCHAR2(10) PRIMARY KEY,
  CREADIT_HOURS NUMBER NOT NULL,
  DEPARTMENT VARCHAR2(20)
);

CREATE TABLE SECTION
(
  SECTION_IDENTIFIER NUMBER PRIMARY KEY,
  COURSE_NUMBER VARCHAR2(10),
  SEMESTER VARCHAR2(10) NOT NULL,
  YEAR NUMBER,
  INSTRUCTOR VARCHAR2(20),
  FOREIGN KEY (COURSE_NUMBER) REFERENCES COURSE(COURSE_NUMBER)
);

CREATE TABLE GRADE_REPORT
(
  STUDENT_NUMBER NUMBER,
  SECTION_IDENTIFIER NUMBER,
  GRADE CHAR(1) NOT NULL,
  PRIMARY KEY (STUDENT_NUMBER,SECTION_IDENTIFIER),
  FOREIGN KEY (STUDENT_NUMBER) REFERENCES SECTION(STUDENT_NUMBER),
  FOREIGN KEY (SECTION_IDENTIFIER) REFERENCE SECTION(SECTION_IDENTIFIER)
);

CREATE TABLE PREREQUISITE
(
  COURSE_NUMBER VARCHAR2(10),
  PREREQUISITE_NUMBER VARCHAR2(10),
  PRIMARY KEY (COURSE_NUMBER,PREREQUISITE_NUMBER),
  FOREIGN KEY (COURSE_NUMBER) REFERENCE COURSE(COURSE_NUMBER)
);

```
