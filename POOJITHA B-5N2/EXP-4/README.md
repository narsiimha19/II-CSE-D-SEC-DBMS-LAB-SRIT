(4) 3. Create STUDENT4 table

(

    SNAME VARCHAR2(30),

);
```

(4) 4. Apply constraints to STUDENT4
```
ALTER TABLE STUDENT4
ADD CONSTRAINT PK_STUDENT4 PRIMARY KEY (SID);
ALTER TABLE STUDENT4
MODIFY SNAME NOT NULL;
ALTER TABLE STUDENT4
ADD CONSTRAINT FK_STUDENT4_DEPT
FOREIGN KEY (DID)
REFERENCES DEPT(DNO);
